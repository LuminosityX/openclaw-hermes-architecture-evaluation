69: def _cached_read(path: Path, cache: dict[str, tuple], parse):
70:     """Shared stat-signature-keyed read; returns a deepcopy of the parsed value.
71: 
72:     ``None`` when the file is absent or fails to parse (fail-open). A parse failure is logged
73:     LOUDLY — the admin needs to know their policy isn't applied — but never raises, so a malformed
74:     managed file can't brick startup.
75:     """
76:     try:
77:         st = path.stat()
78:     except OSError:
79:         return None  # absent
80:     key = file_signature(st)
81:     path_key = str(path)
82:     with _CACHE_LOCK:
83:         hit = cache.get(path_key)
84:         if hit is not None and hit[:len(key)] == key:
85:             return copy.deepcopy(hit[len(key)])
86:     try:
87:         parsed = parse(path)
88:     except Exception as exc:
89:         logger.warning(
90:             "managed scope: failed to parse %s: %s — IGNORING this managed file. "
91:             "Admin policy from this file is NOT being applied. Fix and restart.",
92:             path, exc)
93:         return None
94:     with _CACHE_LOCK:
95:         cache[path_key] = (*key, copy.deepcopy(parsed))
96:     return parsed
